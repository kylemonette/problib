Let $A = \begin{pmatrix} 2 & 1 \\ 1 & 2 \end{pmatrix}$.
\begin{parts}
  \part Find the eigenvalues and eigenvectors of $A$.
  \begin{subparts}
    \subpart[3] Eigenvalues.
    \subpart[3] Eigenvectors.
  \end{subparts}
  \part[4] Diagonalize $A$.
\end{parts}
\begin{solution}
  $\lambda = 1, 3$ with eigenvectors $(1,-1)$ and $(1,1)$.
\end{solution}
\end{problem}
