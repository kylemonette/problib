Let $h(x) = x^2 e^x$.
\begin{parts}
  \part[3] Find $h'(x)$.
  \part[3] Find the equation of the tangent line at $x = 1$.
  \part[4] Find all critical points of $h$.
\end{parts}
\begin{solution}
  $h'(x) = (2x + x^2)e^x$; tangent line $y = 3e(x - 1) + e$; critical points $x = 0, -2$.
\end{solution}
\end{problem}
