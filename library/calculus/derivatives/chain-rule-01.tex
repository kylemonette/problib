Differentiate each function.
\begin{parts}
  \part[4] $f(x) = \sin(x^2)$
  \part[6] $g(x) = \cos^3(2x)$
\end{parts}
\begin{solution}
  (a) $2x\cos(x^2)$ \qquad (b) $-6\cos^2(2x)\sin(2x)$
\end{solution}
\end{problem}
